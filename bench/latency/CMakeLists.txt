# eigenbook_latency: per-operation latency benchmark.
# eigenbook_latency_instrumented: the same, with EIGENBOOK_INSTRUMENT_INDEX (index probe
# statistics; slightly slower index, so its latencies are not comparable with the plain build).

# Build facts the benchmark prints with its results.
set(_eigenbook_git_sha "unknown")
find_package(Git QUIET)
if (GIT_FOUND)
    execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse --short=12 HEAD
            WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
            OUTPUT_VARIABLE _sha OUTPUT_STRIP_TRAILING_WHITESPACE
            RESULT_VARIABLE _rc ERROR_QUIET)
    if (_rc EQUAL 0)
        set(_eigenbook_git_sha "${_sha}")
        execute_process(COMMAND ${GIT_EXECUTABLE} status --porcelain --untracked-files=no
                WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
                OUTPUT_VARIABLE _dirty OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        if (NOT _dirty STREQUAL "")
            string(APPEND _eigenbook_git_sha "-dirty")
        endif ()
    endif ()
endif ()

set(_eigenbook_flags "${CMAKE_CXX_FLAGS}")
foreach (cfg Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER ${cfg} CFG)
    string(APPEND _eigenbook_flags "$<$<CONFIG:${cfg}>: ${CMAKE_CXX_FLAGS_${CFG}}>")
endforeach ()
if (CMAKE_INTERPROCEDURAL_OPTIMIZATION)
    string(APPEND _eigenbook_flags " [LTO]")
endif ()
if (EIGENBOOK_SANITIZE)
    string(APPEND _eigenbook_flags " [sanitize=${EIGENBOOK_SANITIZE}]")
endif ()

file(GENERATE
        OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>/eigenbook_bench_build_info.hpp
        CONTENT "#pragma once
// Generated by bench/latency/CMakeLists.txt. Git state is taken at configure time.
#define EIGENBOOK_BENCH_COMPILER R\"eb(${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION})eb\"
#define EIGENBOOK_BENCH_BUILD_TYPE R\"eb($<CONFIG>)eb\"
#define EIGENBOOK_BENCH_FLAGS R\"eb(${_eigenbook_flags})eb\"
#define EIGENBOOK_BENCH_GIT_SHA R\"eb(${_eigenbook_git_sha})eb\"
#define EIGENBOOK_BENCH_RESULTS_DIR R\"eb(${PROJECT_SOURCE_DIR}/bench/results)eb\"
")

foreach (target eigenbook_latency eigenbook_latency_instrumented)
    add_executable(${target} main.cpp)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>)
    target_link_libraries(${target} PRIVATE eigenbook_bench_support eigenbook_warnings)
    eigenbook_copy_sanitizer_runtime(${target})
endforeach ()
target_compile_definitions(eigenbook_latency_instrumented PRIVATE EIGENBOOK_INSTRUMENT_INDEX)
